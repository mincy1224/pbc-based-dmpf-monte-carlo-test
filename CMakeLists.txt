cmake_minimum_required(VERSION 3.24)
project(cuckoo_bench LANGUAGES CXX)
option(CUCKOO_HOST_TESTS_ONLY "Build checkpoint tests without a CUDA toolkit" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(checkpoint_tests host_tests.cpp)
target_link_libraries(checkpoint_tests PRIVATE Threads::Threads)
add_test(NAME checkpoint_tests COMMAND checkpoint_tests "${CMAKE_CURRENT_BINARY_DIR}/checkpoint_test_data")
if(CUCKOO_HOST_TESTS_ONLY)
    return()
endif()
enable_language(CUDA)
add_executable(cuckoo_bench cuckoo_bench.cu)
target_compile_options(cuckoo_bench PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo>")
target_link_libraries(cuckoo_bench PRIVATE Threads::Threads)
set_target_properties(cuckoo_bench PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
