cmake_minimum_required(VERSION 3.24)
project(cuckoo_bench LANGUAGES CXX)
option(CUCKOO_HOST_TESTS_ONLY "Build checkpoint tests without a CUDA toolkit" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
enable_testing()
add_executable(checkpoint_tests host_tests.cpp)
target_link_libraries(checkpoint_tests PRIVATE Threads::Threads)
add_test(NAME checkpoint_tests COMMAND checkpoint_tests "${CMAKE_CURRENT_BINARY_DIR}/checkpoint_test_data")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64|i[3-6]86)$")
    foreach(cpu_target IN ITEMS cpu_bench cpu_tests gpu_arch_tests)
        add_executable(${cpu_target} ${cpu_target}.cpp)
        target_link_libraries(${cpu_target} PRIVATE Threads::Threads)
        if(MSVC)
            target_compile_options(${cpu_target} PRIVATE /O2)
        else()
            target_compile_options(${cpu_target} PRIVATE -O3 -maes)
        endif()
    endforeach()
    add_test(NAME cpu_aesni_tests COMMAND cpu_tests "${CMAKE_CURRENT_BINARY_DIR}/cpu_test_data")
    set_tests_properties(cpu_aesni_tests PROPERTIES SKIP_RETURN_CODE 77)
    add_test(NAME gpu_arch_host_tests COMMAND gpu_arch_tests)
    set_tests_properties(gpu_arch_host_tests PROPERTIES SKIP_RETURN_CODE 77)
endif()
if(CUCKOO_HOST_TESTS_ONLY)
    return()
endif()
enable_language(CUDA)
add_executable(cuckoo_bench cuckoo_bench.cu)
target_compile_options(cuckoo_bench PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;-lineinfo;--resource-usage>")
target_link_libraries(cuckoo_bench PRIVATE Threads::Threads)
set_target_properties(cuckoo_bench PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
